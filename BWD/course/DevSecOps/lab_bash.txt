#jq is built to take JSON input and emit JSON output, such as prettify JSON object
echo '{"cloudprovider":{"name":"AWS","url":"www.aws.com"}} ' | jq '.'
jq '.' filename.json


#jq '.[] | .details' command to extract details parts of JSON
jq '.[] | .details' filename.json
jq '.[] | {name: .details.name, url:.details.url}' filename.json
jq '.[1] | {name: .details.name, url: .details.url}' filename.json
jq '[.[] | {name: .details.name, url: .details.url}]' filename.json
jq '.[] | {name: .details.name, url: .details.url, services: .services[]}' filename.json
jq '.[] | {name: .details.name, url: .details.url, services: [.services[]]}' filename.json
jq '.[] | {name: .details.name, url: .details.url, services: .services}' filename.json


#jq with length, select, 
jq '.[] | {name: .details.name, servicecount: .services | length}' filename.json
jq '.[] | select(.details.name=="AWS")' filename.json
jq '.[] | select(.services[].type=="CI-CD") | .details.name' filename.json
jq '.[].services[] | select(.type=="CI-CD")' filename.json
jq '.[].services[] | select(.type=="CI-CD") | .name' filename.json

#Sed

#replace banana to mango, s/ search, /g get, it won't edit the actual file
