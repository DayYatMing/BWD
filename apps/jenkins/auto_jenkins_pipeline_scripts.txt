#!/usr/bin/env groovy

def SendEmail(stageName, status, c) {
    def url = "https://jenkins.local.bw-digital.com/job/${env.JOB_NAME}/${env.BUILD_NUMBER}/"
    def url_proceed = url + "input/Input/proceedEmpty"
    def url_abort = url + "input/Input/abort"
    def str = ""
    def to = "oyip@bw-digital.com"

    if(c == "1") {
        str = "<p>Do you want to <a href='${url_proceed}'>proceed</a> or <a href='${url_abort}'>abort</a> the current stage?</p>"
    }

    emailext (
        recipientProviders: [[$class: 'DevelopersRecipientProvider']],
        subject: "Jenkins Pipeline - ${env.JOB_NAME} [#${env.BUILD_NUMBER}]",
        body: "<p>Stage: " + stageName + "</p>" +
            "<p>Status: <b>" + status.toUpperCase() + "</b></p>"+
            "<p>Job: ${env.JOB_NAME} [#${env.BUILD_NUMBER}]</p>" +
            str ,
        to: "${to}"
    )
}

node {
    def pauseMsg = "Continue to Prod build?"
    def userInput = ""

    stage('Jenkins Pipeline Auto (Pre-prod)') {
        echo env.STAGE_NAME + ' - Start.'
        SendEmail(env.STAGE_NAME, "Start", "")
        build job: 'Jenkins_Pipeline_Test'
    }

    stage('Proceeding from Pre-prod to Prod') {
        echo env.STAGE_NAME + ' - Pause'

        try{
            SendEmail(env.STAGE_NAME, "Pause", "1")
            input(id: 'input', message: pauseMsg)
            userInput = "yes"

            checkout scm
            echo "-----------------"
            sh "git merge origin/pipeline_dev"
            echo "================="

        }catch(err){
            SendEmail(env.STAGE_NAME, "Cancel", "")
            userInput = "no"
        }
    }

    stage('Jenkins Pipeline Auto (Prod)') {
        if(userInput != "yes"){
            error ("Prod build is cancelled, and this pipeline is stopped.")
        }

        echo env.STAGE_NAME + ' - Start'
        SendEmail(env.STAGE_NAME, "Start", "")
        build job: 'Jenkins_Pipeline_Prod'
    }
}

    )
}

    def hosts = []

    stage('Preliminary settings (Pre-prod)') {
        if (params.Host == 'All') {
            hosts = ['10.7.1.5','10.10.1.4']
        } else {
            hosts = [params.Host]
        }
        deploymentType = params.Deployment
        echo "Host: ${hosts}"
        echo "Deployment: ${deploymentType}"

        checkout scm
    }

    stage('Snyk scanning (Pre-prod)') {
        echo 'use Snyk to check vulnerabilities and security issues'
    }

    stage('Docker packaging (Pre-prod)') {
        echo 'packaging the project into a deployable .war file'
    }

    stage('Configuring (Pre-prod)') {
        echo 'config the docker container for the project'
    }

    stage('Server starting (Pre-prod)') {
        echo 'start container with the project'

        try{
            //error("@" + env.STAGE_NAME)
        }catch(err){
            SendEmail(env.STAGE_NAME, "Failed", hosts, deploymentType, err)
            error("Process stopped due to step failure.")
        }

    stage('Deployment checking (Pre-prod)') {
        echo 'after deployment check'
    }
